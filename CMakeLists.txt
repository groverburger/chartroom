cmake_minimum_required(VERSION 3.20)
project(Chartroom VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(chartroom_core src/core.cpp src/drawings.cpp src/market.cpp)
target_include_directories(chartroom_core PUBLIC src vendor/json)
if(MSVC)
  target_compile_options(chartroom_core PRIVATE /W4)
else()
  target_compile_options(chartroom_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  add_executable(chartroom_market_tests test/market_test.cpp)
  target_link_libraries(chartroom_market_tests PRIVATE chartroom_core)
  add_test(NAME market_providers COMMAND chartroom_market_tests "${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  add_executable(chartroom_tests test/core_test.cpp)
  target_link_libraries(chartroom_tests PRIVATE chartroom_core)
  add_test(NAME core COMMAND chartroom_tests)
  add_executable(chartroom_drawing_tests test/drawing_test.cpp)
  target_link_libraries(chartroom_drawing_tests PRIVATE chartroom_core)
  add_test(NAME drawings COMMAND chartroom_drawing_tests)
  add_executable(chartroom_vix_tests test/vix_test.cpp)
  target_link_libraries(chartroom_vix_tests PRIVATE chartroom_core)
  add_test(NAME vix_recovery COMMAND chartroom_vix_tests "${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
endif()
option(CHARTROOM_CORE_ONLY "Build only the model and tests" OFF)
if(CHARTROOM_CORE_ONLY)
  return()
endif()
if(EMSCRIPTEN)
  set(CHARTROOM_PLATFORM web)
else()
  set(CHARTROOM_PLATFORM "${CMAKE_SYSTEM_NAME}")
endif()
add_custom_target(chartroom_build_stamp
  COMMAND ${CMAKE_COMMAND} "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}" "-DBINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}"
    "-DVERSION=${PROJECT_VERSION}" "-DPLATFORM=${CHARTROOM_PLATFORM}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/BuildStamp.cmake"
  BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/generated/build_version.hpp"
  VERBATIM)
set(IMGUI vendor/imgui)
add_library(imgui ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_glfw.cpp ${IMGUI}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
# Long histories can exceed 64K vertices, including on WebGL2.
target_compile_definitions(imgui PUBLIC "ImDrawIdx=unsigned int")
add_executable(chartroom MACOSX_BUNDLE src/main.cpp src/render_schedule.cpp src/app.cpp src/drawing_ui.cpp src/net.cpp src/state.cpp src/providers.cpp src/market_ui.cpp)
add_dependencies(chartroom chartroom_build_stamp)
target_link_libraries(chartroom PRIVATE chartroom_core imgui)
target_compile_definitions(chartroom PRIVATE GL_SILENCE_DEPRECATION)
if(EMSCRIPTEN)
  target_compile_options(chartroom_core PUBLIC -fexceptions)
  target_compile_definitions(imgui PUBLIC IMGUI_IMPL_OPENGL_ES3)
  target_link_options(chartroom PRIVATE -fexceptions -sUSE_GLFW=3 -sFULL_ES3=1 -sFETCH=1 -sALLOW_MEMORY_GROWTH=1 -sFORCE_FILESYSTEM=1 -lidbfs.js "SHELL:--pre-js ${CMAKE_CURRENT_SOURCE_DIR}/web/storage.js" "SHELL:--pre-js ${CMAKE_CURRENT_SOURCE_DIR}/web/scheduler.js" --shell-file ${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html)
  set_target_properties(chartroom PROPERTIES SUFFIX .html)
  set_property(TARGET chartroom APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/web/storage.js" "${CMAKE_CURRENT_SOURCE_DIR}/web/scheduler.js" "${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html")
else()
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/glfw EXCLUDE_FROM_ALL)
  find_package(OpenGL REQUIRED)
  find_package(CURL REQUIRED)
  find_package(Threads REQUIRED)
  target_link_libraries(imgui PUBLIC glfw OpenGL::GL)
  target_link_libraries(chartroom PRIVATE CURL::libcurl Threads::Threads)
  set_target_properties(chartroom PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.chartroom.cpp MACOSX_BUNDLE_BUNDLE_NAME Chartroom MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION})
  # App icons, rendered from assets/icon/chartroom.svg by scripts/make_icons.py.
  if(APPLE)
    set(CHARTROOM_ICNS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/Chartroom.icns")
    target_sources(chartroom PRIVATE "${CHARTROOM_ICNS}")
    set_source_files_properties("${CHARTROOM_ICNS}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(chartroom PROPERTIES MACOSX_BUNDLE_ICON_FILE Chartroom.icns)
  elseif(WIN32)
    target_sources(chartroom PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/chartroom.rc")
    # Launch without a console window while keeping main() as the entry point.
    if(MSVC)
      set_target_properties(chartroom PROPERTIES WIN32_EXECUTABLE ON)
      target_link_options(chartroom PRIVATE /ENTRY:mainCRTStartup)
    endif()
  endif()
  install(TARGETS chartroom BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
# Embed the font so launch location and app-bundle resources cannot break the UI.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/Roboto-Regular.ttf" FONT_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," FONT_BYTES "${FONT_HEX}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/font.hpp" "#pragma once\ninline unsigned char chartroom_font[]={${FONT_BYTES}};\n")
target_include_directories(chartroom PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  add_executable(chartroom_state_tests test/state_test.cpp src/state.cpp src/net.cpp src/providers.cpp)
  target_link_libraries(chartroom_state_tests PRIVATE chartroom_core CURL::libcurl Threads::Threads)
  add_test(NAME persistence COMMAND chartroom_state_tests)
  add_executable(chartroom_gui_tests test/gui_test.cpp src/render_schedule.cpp src/app.cpp src/drawing_ui.cpp src/state.cpp src/net.cpp src/providers.cpp src/market_ui.cpp)
  add_dependencies(chartroom_gui_tests chartroom_build_stamp)
  target_include_directories(chartroom_gui_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(chartroom_gui_tests PRIVATE chartroom_core imgui CURL::libcurl Threads::Threads)
  target_compile_definitions(chartroom_gui_tests PRIVATE GL_SILENCE_DEPRECATION)
endif()
if(APPLE AND NOT EMSCRIPTEN)
  add_custom_command(TARGET chartroom POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_CONTENT_DIR:chartroom>/Resources/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md"
      "${CMAKE_CURRENT_SOURCE_DIR}/assets/Roboto-LICENSE.txt"
      "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui/LICENSE.txt"
      "${CMAKE_CURRENT_SOURCE_DIR}/vendor/glfw/LICENSE.md"
      "${CMAKE_CURRENT_SOURCE_DIR}/vendor/json/LICENSE.MIT"
      "${CMAKE_CURRENT_SOURCE_DIR}/vendor/OpenTerminal-LICENSE.txt"
      "$<TARGET_BUNDLE_CONTENT_DIR:chartroom>/Resources/licenses")
endif()

if(APPLE AND NOT EMSCRIPTEN)
  set(CHARTROOM_MANIFEST_DIR "$<TARGET_BUNDLE_CONTENT_DIR:chartroom>/Resources")
else()
  set(CHARTROOM_MANIFEST_DIR "$<TARGET_FILE_DIR:chartroom>")
endif()
add_custom_command(TARGET chartroom POST_BUILD
  COMMAND ${CMAKE_COMMAND} "-DBINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}" "-DAPP_DIR=${CHARTROOM_MANIFEST_DIR}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/PublishBuild.cmake" VERBATIM)
if(NOT EMSCRIPTEN AND NOT APPLE)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build-info.json" DESTINATION bin)
endif()
